@section Scripts {
    <!-- jQuery is required for the client-side validation and step navigation -->
    <script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.0.0-beta3/js/all.min.js" crossorigin="anonymous"></script>

    @{
        await Html.RenderPartialAsync("_ValidationScriptsPartial");
    }

    <script>
        $(document).ready(function () {
            const form = $('#profileForm');
            const step1 = $('#step1');
            const step2 = $('#step2');
            const nextBtn = $('#nextBtn');
            const backBtn = $('#backBtn');
            const submitBtn = $('#submitBtn');
            const modalTitle = $('#modalTitle');

            // Set initial title
            modalTitle.text('Client Profile Setup (Step 1 of 2)');

            /**
             * Navigates from Step 1 to Step 2 after validating Step 1 fields.
             */
            nextBtn.on('click', function (e) {
                e.preventDefault();
                
                // Get all required inputs in step 1
                const step1Inputs = step1.find('input[required], select[required], textarea[required]').get();
                let allStep1Valid = true;
                
                // Manually trigger validation for each required field in step 1.
                // This ensures validation messages appear for only step 1 fields.
                step1Inputs.forEach(input => {
                    // form.validate().element(input) checks the validity of a single element
                    if (!form.validate().element(input)) {
                        allStep1Valid = false;
                    }
                });

                if (allStep1Valid) {
                    // Hide Step 1, Show Step 2
                    step1.addClass('d-none');
                    step2.removeClass('d-none');

                    // Update Buttons
                    nextBtn.addClass('d-none');
                    backBtn.removeClass('d-none');
                    submitBtn.removeClass('d-none');
                    
                    // Update title
                    modalTitle.text('Client Profile Setup (Step 2 of 2)');
                }
            });

            /**
             * Navigates back from Step 2 to Step 1.
             */
            backBtn.on('click', function () {
                // Hide Step 2, Show Step 1
                step2.addClass('d-none');
                step1.removeClass('d-none');

                // Update Buttons
                submitBtn.addClass('d-none');
                backBtn.addClass('d-none');
                nextBtn.removeClass('d-none');
                
                // Update title
                modalTitle.text('Client Profile Setup (Step 1 of 2)');
            });
            
            // Re-validate the entire form on submission of step 2
            form.on('submit', function() {
                // This ensures that all hidden fields (even if they were validated in step 1) 
                // and the newly visible fields in step 2 are checked before POST.
                return form.valid();
            });
        });
    </script>
}
